import React, { useState } from 'react'
import { motion } from 'framer-motion'
import { Eye, EyeOff, LogIn, Mail, Lock, Google, Facebook } from 'lucide-react'

// Single-file React component for a polished login page
// - TailwindCSS utility classes assumed available
// - Uses Framer Motion for subtle animations
// - Replace OAuth URLs and /api endpoints with your backend

export default function LoginPage() {
  const [email, setEmail] = useState('')
  const [password, setPassword] = useState('')
  const [showPassword, setShowPassword] = useState(false)
  const [remember, setRemember] = useState(false)
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState('')

  const validate = () => {
    if (!email) return 'Email is required.'
    // simple email regex
    const re = /^\S+@\S+\.\S+$/
    if (!re.test(email)) return 'Enter a valid email.'
    if (!password) return 'Password is required.'
    if (password.length < 6) return 'Password must be 6+ characters.'
    return ''
  }

  const handleSubmit = async (e) => {
    e.preventDefault()
    const v = validate()
    if (v) return setError(v)
    setError('')
    setLoading(true)
    try {
      // Example: POST to your auth endpoint
      const res = await fetch('/api/auth/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email, password, remember }),
      })
      const data = await res.json()
      if (!res.ok) throw new Error(data?.message || 'Login failed')

      // Handle flows:
      // - If 2FA required, redirect to /2fa
      // - If success, redirect to dashboard
      if (data?.requires2FA) {
        window.location.href = '/auth/2fa'
        return
      }
      window.location.href = data?.redirectTo || '/dashboard'
    } catch (err) {
      setError(err.message)
    } finally {
      setLoading(false)
    }
  }

  const oauthSignIn = (provider) => {
    // Replace with your real OAuth endpoints
    // e.g., /api/auth/oauth/google or external provider URL
    window.location.href = `/api/auth/oauth/${provider}`
  }

  return (
    <div className="min-h-screen flex items-center justify-center bg-gradient-to-b from-slate-900 via-purple-900 to-black px-4">
      <motion.div
        initial={{ opacity: 0, y: 12 }}
        animate={{ opacity: 1, y: 0 }}
        transition={{ duration: 0.45 }}
        className="w-full max-w-md bg-[#0b1020]/80 backdrop-blur-md rounded-2xl shadow-2xl p-8 border border-white/5"
      >
        <header className="flex items-center gap-4 mb-6">
          <div className="w-14 h-14 rounded-lg bg-gradient-to-br from-cyan-400 to-indigo-500 flex items-center justify-center shadow-lg">
            {/* Small orb logo */}
            <svg width="34" height="34" viewBox="0 0 24 24" fill="none" xmlns="http://www.w3.org/2000/svg">
              <circle cx="12" cy="12" r="8" fill="#7dd3fc" fillOpacity="0.12" />
              <circle cx="12" cy="12" r="5" fill="#60a5fa" />
            </svg>
          </div>
          <div>
            <h1 className="text-2xl font-semibold text-white tracking-tight">Nexus Mana</h1>
            <p className="text-sm text-slate-300 mt-0.5">Sign in to your community dashboard</p>
          </div>
        </header>

        <form onSubmit={handleSubmit} className="space-y-4">
          {error && <div className="text-sm text-red-400 bg-red-900/10 p-2 rounded">{error}</div>}

          <label className="block">
            <span className="text-xs text-slate-300">Email</span>
            <div className="mt-1 relative">
              <input
                type="email"
                value={email}
                onChange={(e) => setEmail(e.target.value)}
                placeholder="you@domain.com"
                className="w-full pl-10 pr-3 py-2 rounded-md bg-white/3 border border-white/5 text-white placeholder:text-slate-400 focus:outline-none focus:ring-2 focus:ring-cyan-400"
              />
              <div className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-300">
                <Mail size={16} />
              </div>
            </div>
          </label>

          <label className="block">
            <div className="flex items-center justify-between">
              <span className="text-xs text-slate-300">Password</span>
              <a href="/auth/forgot" className="text-xs text-cyan-300 hover:underline">Forgot?</a>
            </div>
            <div className="mt-1 relative">
              <input
                type={showPassword ? 'text' : 'password'}
                value={password}
                onChange={(e) => setPassword(e.target.value)}
                placeholder="••••••••"
                className="w-full pl-10 pr-10 py-2 rounded-md bg-white/3 border border-white/5 text-white placeholder:text-slate-400 focus:outline-none focus:ring-2 focus:ring-cyan-400"
              />
              <div className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-300">
                <Lock size={16} />
              </div>
              <button
                type="button"
                onClick={() => setShowPassword((s) => !s)}
                className="absolute right-2 top-1/2 -translate-y-1/2 text-slate-300 p-1 rounded"
                aria-label={showPassword ? 'Hide password' : 'Show password'}
              >
                {showPassword ? <EyeOff size={16} /> : <Eye size={16} />}
              </button>
            </div>
          </label>

          <div className="flex items-center justify-between">
            <label className="inline-flex items-center gap-2 text-sm text-slate-300">
              <input type="checkbox" checked={remember} onChange={() => setRemember(r => !r)} className="rounded bg-white/5" />
              Remember me
            </label>
            <a href="/auth/help" className="text-sm text-slate-400 hover:text-cyan-300">Help</a>
          </div>

          <button
            type="submit"
            disabled={loading}
            className="w-full mt-1 inline-flex items-center justify-center gap-2 py-2 rounded-lg bg-gradient-to-r from-cyan-400 to-indigo-500 text-black font-semibold shadow-md hover:brightness-105 disabled:opacity-60"
          >
            {loading ? 'Signing in...' : 'Sign in'} <LogIn size={16} />
          </button>
        </form>

        <div className="my-4 flex items-center gap-3 text-slate-400">
          <span className="flex-1 h-px bg-white/5" />
          <span className="text-xs">Or continue with</span>
          <span className="flex-1 h-px bg-white/5" />
        </div>

        <div className="grid grid-cols-2 gap-3">
          <button onClick={() => oauthSignIn('google')} className="py-2 rounded-lg border border-white/6 bg-white/5 flex items-center justify-center gap-2 text-sm text-white hover:bg-white/6">
            <svg width="16" height="16" viewBox="0 0 24 24" fill="none"><path d="M21.35 11.1h-9.2v2.98h5.28c-.23 1.46-1.42 4.28-5.28 4.28-3.18 0-5.78-2.62-5.78-5.86S8.97 6.65 12.15 6.65c1.81 0 3.02.78 3.72 1.44l2.56-2.47C17.3 3.6 14.99 2.5 12.15 2.5 6.8 2.5 2.5 6.85 2.5 12s4.3 9.5 9.65 9.5c5.58 0 9.35-3.92 9.35-9.4 0-.64-.06-1.12-.15-1.5z" fill="#fff"/></svg>
            Google
          </button>
          <button onClick={() => oauthSignIn('facebook')} className="py-2 rounded-lg border border-white/6 bg-white/5 flex items-center justify-center gap-2 text-sm text-white hover:bg-white/6">
            <svg width="14" height="14" viewBox="0 0 24 24" fill="none"><path d="M15.12 8.5h1.9V5.44h-1.9c-.97 0-1.86.79-1.86 1.87v1.19H12v2.4h2.15V20h2.58v-8.92h1.82l.29-2.4h-2.11V9.87c0-.42.1-.7.77-.7z" fill="#fff"/></svg>
            Facebook
          </button>
        </div>

        <p className="text-center text-sm text-slate-400 mt-6">
          Don’t have an account? <a href="/auth/register" className="text-cyan-300 hover:underline">Create one</a>
        </p>

        <footer className="mt-6 text-xs text-slate-500 text-center">
          By signing in you agree to our <a href="/terms" className="text-slate-300 underline">Terms</a> and <a href="/privacy" className="text-slate-300 underline">Privacy Policy</a>
        </footer>
      </motion.div>
    </div>
  )
}

/*
  Quick backend notes:
  - POST /api/auth/login { email, password, remember } -> returns { ok, requires2FA?, redirectTo }
  - GET /api/auth/oauth/:provider -> starts OAuth flow (Google, Facebook, Discord, etc.)
  - Implement rate-limiting, lockouts, and IP logging on the server side
  - Use secure cookies (HttpOnly, Secure, SameSite) for sessions
  - Serve over HTTPS and add 2FA (TOTP / SMS) support for extra security

  How to use:
  - Add this component into a React app that has Tailwind configured.
  - Replace fetch endpoints with your real API routes.
  - Customize colors, logos, and copy to match your brand.
*/
